// Copy & paste this into Google Apps Script (Code.gs):

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // Headers check
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Student Name", "Class", "Score", "Clues Found", "Accuracy (%)", "Time Spent (s)"]);
  }
  
  sheet.appendRow([
    new Date(),
    data.studentName,
    data.studentClass,
    data.score,
    data.cluesFound,
    data.accuracy,
    data.timeTaken
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"result": "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
